fkptjax vs sMGPT cross-check -- Wolfram Cloud snippet
======================================================

How to use:
1. Go to cloud.wolframcloud.com, sign in or create a free Wolfram ID.
2. Click "Create New" -> "Notebook".
3. Paste everything below the line of ='s into a single input cell.
4. Press Shift+Enter to evaluate (should take well under a minute).
5. Copy ALL the printed output (starting from "=== Part 1 ===") and send
   it back to Claude.

This is self-contained -- no external files needed. It evaluates sMGPT's
own equations (2_P22type.wl / 3_P13type.wl) directly, for a BZ_Mass-like
toy MG model, at a handful of hand-picked test points, so the numbers can
be diffed against fkptjax's Python/JAX implementation at the same inputs.

(* ===================== fkptjax cross-check snippet ===================== *)
om = 0.3; lambdaA = 0.5; lambdaDS = 0.8; muKinf = 1.3;

mu[eta_, k_] := Module[{a, Dd, X},
   a = Exp[eta]; Dd = lambdaDS*a^(-3) + lambdaA;
   X = (k*lambdaA*lambdaDS/(Dd*a))^2; (1 + muKinf*X)/(1 + X)];

f1[eta_] := 2. - 3./(2.(1. + (1. - om)/om Exp[3. eta]));
f2[eta_] := 3./(2.(1. + (1. - om)/om Exp[3. eta]));
etaini = -3.912023; etaev = Log[1./1.5];

growthF[k_?NumericQ] := Module[{sol},
  sol = NDSolve[{Dv''[eta] + f1[eta] Dv'[eta] - f2[eta] mu[eta, k] Dv[eta] == 0,
     Dv[etaini] == Exp[etaini], Dv'[etaini] == Exp[etaini]}, Dv, {eta, etaini, etaev},
    MaxSteps -> Infinity, AccuracyGoal -> 12, PrecisionGoal -> 12];
  {Dv[etaev], Dv'[etaev]} /. sol[[1]]];

Print["=== Part 1: linear growth ==="];
{Dk01, dDk01} = growthF[0.1]; fk01 = dDk01/Dk01; Print["f(k=0.1)        = ", fk01];
{Dk008, dDk008} = growthF[0.08]; fk008 = dDk008/Dk008; Print["f(k=0.08)       = ", fk008];
{Dkm, dDkm} = growthF[0.08246211]; fkm = dDkm/Dkm; Print["f(kminus=0.0825) = ", fkm];
{Dk0, dDk0} = growthF[0.000001]; f0 = dDk0/Dk0; Print["f0 (k->0)       = ", f0];

sourceA[kf_, k1_, k2_, eta_] := f2[eta]*(mu[eta, kf] +
    (mu[eta, kf] - mu[eta, k1])*(kf^2 - k1^2 - k2^2)/(2*k2^2) +
    (mu[eta, kf] - mu[eta, k2])*(kf^2 - k1^2 - k2^2)/(2*k1^2));
sourceb[kf_, k1_, k2_, eta_] := f2[eta]*(mu[eta, k1] + mu[eta, k2] - mu[eta, kf]);
D2plusi = 3. Exp[2. etaini]/7.; dD2plusi = 6. Exp[2. etaini]/7.;

solveAB[kf_?NumericQ, k1_?NumericQ, k2_?NumericQ] := Module[
  {sol, Dpk1v, dDpk1v, Dpk2v, dDpk2v, Afv, dAfv, Bfv, dBfv, C, Cp},
  sol = NDSolve[{
     Dpk1''[eta] + f1[eta] Dpk1'[eta] - f2[eta] mu[eta, k1] Dpk1[eta] == 0,
     Dpk2''[eta] + f1[eta] Dpk2'[eta] - f2[eta] mu[eta, k2] Dpk2[eta] == 0,
     Af''[eta] + f1[eta] Af'[eta] - f2[eta] mu[eta, kf] Af[eta] == sourceA[kf, k1, k2, eta] Dpk1[eta] Dpk2[eta],
     Bf''[eta] + f1[eta] Bf'[eta] - f2[eta] mu[eta, kf] Bf[eta] == sourceb[kf, k1, k2, eta] Dpk1[eta] Dpk2[eta],
     Dpk1[etaini] == Exp[etaini], Dpk1'[etaini] == Exp[etaini],
     Dpk2[etaini] == Exp[etaini], Dpk2'[etaini] == Exp[etaini],
     Af[etaini] == D2plusi, Af'[etaini] == dD2plusi,
     Bf[etaini] == D2plusi, Bf'[etaini] == dD2plusi},
    {Dpk1, Dpk2, Af, Bf}, {eta, etaini, etaev}, MaxSteps -> Infinity, AccuracyGoal -> 12, PrecisionGoal -> 12];
  Dpk1v = Dpk1[etaev] /. sol[[1]]; dDpk1v = Dpk1'[etaev] /. sol[[1]];
  Dpk2v = Dpk2[etaev] /. sol[[1]]; dDpk2v = Dpk2'[etaev] /. sol[[1]];
  Afv = Af[etaev] /. sol[[1]]; dAfv = Af'[etaev] /. sol[[1]];
  Bfv = Bf[etaev] /. sol[[1]]; dBfv = Bf'[etaev] /. sol[[1]];
  C = (3./7.) Dpk1v Dpk2v; Cp = (3./7.) (dDpk1v Dpk2v + Dpk1v dDpk2v);
  {Afv/C, Bfv/C, dAfv/C - Afv Cp/C^2, dBfv/C - Bfv Cp/C^2}];

Print[""]; Print["=== Part 2: A/B pair at (kf,k1,k2)=(0.1,0.05,0.08) ==="];
{A1, B1, Ap1, Bp1} = solveAB[0.1, 0.05, 0.08];
Print["A=", A1, "  B=", B1, "  Ap=", Ap1, "  Bp=", Bp1];

Print[""]; Print["=== Part 3: A/B pair, N ordering, (kf,k1,k2)=(0.08,0.1,0.08246211) ==="];
{AN, BN, ApN, BpN} = solveAB[0.08, 0.1, 0.08246211];
Print["A_N=", AN, "  B_N=", BN, "  Ap_N=", ApN, "  Bp_N=", BpN];

AngleEvN[r_, x_] := -((1 - r x)/Sqrt[1 + r^2 - 2. r x]);
F2evN[r_, x_, A_, B_] := 0.5 + 3./14. A + (0.5 - 3./14. B) AngleEvN[r, x]^2 +
   AngleEvN[r, x]/2 (Sqrt[1 + r^2 - 2. r x] + 1/Sqrt[1 + r^2 - 2. r x]);
G2evN[r_, x_, A_, B_, Aprime_, Bprime_, fk_, fkminusp_] :=
  3./14. A (fk + fkminusp)/f0 + 3/14. Aprime/f0 +
   (1./2 (fk + fkminusp) - 3./14. B (fk + fkminusp) - 3/14. Bprime)*AngleEvN[r, x]^2/f0 +
   AngleEvN[r, x]/(2 f0) (fkminusp Sqrt[1 + r^2 - 2. r x] + fk/Sqrt[1 + r^2 - 2. r x]);

rN = 0.08246211/0.1; xN = 0.6;
f2evNval = F2evN[rN, xN, AN, BN];
g2evNval = G2evN[rN, xN, AN, BN, ApN, BpN, fk01, fkm];
A11val = 2 (g2evNval xN rN + f2evNval rN^2 (1 - xN rN)/(1 + rN^2 - 2 rN xN) fkm/f0);
A12val = -((rN^2 (1 - xN^2))/(1 + rN^2 - 2 rN xN)) g2evNval fkm/f0;
A22val = ((rN^2 (1 - 3 xN^2) + 2 xN rN)/(1 + rN^2 - 2 rN xN) fkm/f0 + 2 xN rN fk01/f0) g2evNval +
    (2 rN^2 (1 - rN xN))/(1 + rN^2 - 2 rN xN) (fkm fk01)/f0^2 f2evNval;
A23val = (rN^2 (xN^2 - 1))/(1 + rN^2 - 2 rN xN) g2evNval (fkm fk01)/f0^2;
A33val = (rN^2 (1 - 3 xN^2) + 2 rN xN)/(1 + rN^2 - 2 rN xN) g2evNval (fkm fk01)/f0^2;

Print[""]; Print["=== Part 4: N-ordering kernels, r=", rN, ", x=", xN, " ==="];
Print["F2evN=", f2evNval, "  G2evN=", g2evNval];
Print["A11=", A11val, "  A12=", A12val, "  A22=", A22val, "  A23=", A23val, "  A33=", A33val];

sourceD2b[eta_, x_, k_, p_] := Module[{kp = Sqrt[k^2 + p^2 + 2 k p x]},
   f2[eta] mu[eta, kp] - f2[eta] (mu[eta, k] + mu[eta, p] - mu[eta, kp]) x^2 +
    f2[eta] (mu[eta, kp]*(p/k + k/p)*x - k x/p mu[eta, k] - p x/k mu[eta, p])];

solveD2fused[x_?NumericQ, k_?NumericQ, p_?NumericQ] := Module[
  {sol, Dpkv, dDpkv, Dppv, dDppv, D2v, dD2v, C, Cp, kplusp},
  kplusp = Sqrt[k^2 + p^2 + 2 k p x];
  sol = NDSolve[{
     Dpk''[eta] + f1[eta] Dpk'[eta] - f2[eta] mu[eta, k] Dpk[eta] == 0,
     Dpp''[eta] + f1[eta] Dpp'[eta] - f2[eta] mu[eta, p] Dpp[eta] == 0,
     D2f''[eta] + f1[eta] D2f'[eta] - f2[eta] mu[eta, kplusp] D2f[eta] == sourceD2b[eta, x, k, p] Dpk[eta] Dpp[eta],
     Dpk[etaini] == Exp[etaini], Dpk'[etaini] == Exp[etaini],
     Dpp[etaini] == Exp[etaini], Dpp'[etaini] == Exp[etaini],
     D2f[etaini] == 3. Exp[2. etaini]/7. (1 - x^2), D2f'[etaini] == 6. Exp[2. etaini]/7. (1 - x^2)},
    {Dpk, Dpp, D2f}, {eta, etaini, etaev}, MaxSteps -> Infinity, AccuracyGoal -> 12, PrecisionGoal -> 12];
  Dpkv = Dpk[etaev] /. sol[[1]]; dDpkv = Dpk'[etaev] /. sol[[1]];
  Dppv = Dpp[etaev] /. sol[[1]]; dDppv = Dpp'[etaev] /. sol[[1]];
  D2v = D2f[etaev] /. sol[[1]]; dD2v = D2f'[etaev] /. sol[[1]];
  C = (3./7.) Dpkv Dppv; Cp = (3./7.) (dDpkv Dppv + Dpkv dDppv);
  {D2v/C, dD2v/C - D2v Cp/C^2}];

Print[""]; Print["=== Part 5: fused D2 (AminusBx2) at x=-0.6,k=0.1,p=0.08 ==="];
{gamma2evR, gamma2evRp} = solveD2fused[-0.6, 0.1, 0.08];
Print["AminusBx2=", gamma2evR, "  AminusBx2Prime=", gamma2evRp];

kppf[x_, k_, p_] := Sqrt[k^2 + p^2 + 2. k p x];
src2a[eta_, x_, k_, p_] := f2[eta] mu[eta, kppf[x, k, p]];
src2b[eta_, x_, k_, p_] := f2[eta] (mu[eta, k] + mu[eta, p] - mu[eta, kppf[x, k, p]]);
src2FL[eta_, x_, k_, p_] := f2[eta] (mu[eta, kppf[x, k, p]]*(p/k + k/p)*x - k x/p mu[eta, k] - p x/k mu[eta, p]);
srcD2[eta_, x_, k_, p_] := src2a[eta, x, k, p] - src2b[eta, x, k, p] x^2 + src2FL[eta, x, k, p];

src3I[eta_, x_, k_, p_, D2fv_, D2mv_, Dpkv_, Dppv_] :=
  (f2[eta] (mu[eta, p] + mu[eta, kppf[x, k, p]] - mu[eta, k]) D2fv Dppv +
      srcD2[eta, x, k, p] Dpkv Dppv Dppv) (1. - x^2)/(1. + (p/k)^2 + 2 p/k x) +
   (f2[eta] (mu[eta, p] + mu[eta, kppf[-x, k, p]] - mu[eta, k]) D2mv Dppv +
      srcD2[eta, -x, k, p] Dpkv Dppv Dppv) (1. - x^2)/(1. + (p/k)^2 - 2 p/k x);

src3IIplus[eta_, x_, k_, p_, D2fv_, Dpkv_, Dppv_] :=
  -f2[eta] (mu[eta, p] + mu[eta, kppf[x, k, p]] - 2 mu[eta, k]) Dppv (D2fv + Dpkv Dppv x^2) -
   f2[eta] (mu[eta, kppf[x, k, p]] - mu[eta, k]) Dpkv Dppv Dppv -
   f2[eta] ((mu[eta, kppf[x, k, p]]*(p/k + k/p)*x - k x/p mu[eta, k] - p x/k mu[eta, p])) Dpkv Dppv Dppv;
src3II[eta_, x_, k_, p_, D2fv_, D2mv_, Dpkv_, Dppv_] :=
  src3IIplus[eta, x, k, p, D2fv, Dpkv, Dppv] + src3IIplus[eta, -x, k, p, D2mv, Dpkv, Dppv];

srcFL3Plus[eta_, x_, k_, p_, D2fv_, Dpkv_, Dppv_] :=
  f2[eta] (((p^2 + p k x)/(k^2 + p^2 + 2. p k x)) (mu[eta, p] - mu[eta, k]) D2fv Dppv +
      ((p^2 + p k x)/p^2) (mu[eta, kppf[x, k, p]] - mu[eta, k]) (D2fv Dppv + (1 + x^2) Dpkv Dppv Dppv) +
      (((p^2 + k^2) x^2)/p^2 + ((p^2 + k^2) x)/(p k)) (mu[eta, kppf[x, k, p]] - mu[eta, k]) Dpkv Dppv Dppv);
srcFL3[eta_, x_, k_, p_, D2fv_, D2mv_, Dpkv_, Dppv_] :=
  srcFL3Plus[eta, x, k, p, D2fv, Dpkv, Dppv] + srcFL3Plus[eta, -x, k, p, D2mv, Dpkv, Dppv];

solveD3fused[x_?NumericQ, k_?NumericQ, p_?NumericQ] := Module[
  {sol, Dpkv, dDpkv, Dppv, dDppv, D3v, dD3v, C, xoneMx2},
  xoneMx2 = 1 - x^2;
  sol = NDSolve[{
     Dpk''[eta] + f1[eta] Dpk'[eta] - f2[eta] mu[eta, k] Dpk[eta] == 0,
     Dpp''[eta] + f1[eta] Dpp'[eta] - f2[eta] mu[eta, p] Dpp[eta] == 0,
     D2f''[eta] + f1[eta] D2f'[eta] - f2[eta] mu[eta, kppf[x, k, p]] D2f[eta] == srcD2[eta, x, k, p] Dpk[eta] Dpp[eta],
     D2m''[eta] + f1[eta] D2m'[eta] - f2[eta] mu[eta, kppf[-x, k, p]] D2m[eta] == srcD2[eta, -x, k, p] Dpk[eta] Dpp[eta],
     D3s''[eta] + f1[eta] D3s'[eta] - f2[eta] mu[eta, k] D3s[eta] ==
        src3I[eta, x, k, p, D2f[eta], D2m[eta], Dpk[eta], Dpp[eta]] +
        src3II[eta, x, k, p, D2f[eta], D2m[eta], Dpk[eta], Dpp[eta]] +
        srcFL3[eta, x, k, p, D2f[eta], D2m[eta], Dpk[eta], Dpp[eta]],
     Dpk[etaini] == Exp[etaini], Dpk'[etaini] == Exp[etaini],
     Dpp[etaini] == Exp[etaini], Dpp'[etaini] == Exp[etaini],
     D2f[etaini] == 3. Exp[2. etaini]/7. xoneMx2, D2f'[etaini] == 6. Exp[2. etaini]/7. xoneMx2,
     D2m[etaini] == 3. Exp[2. etaini]/7. xoneMx2, D2m'[etaini] == 6. Exp[2. etaini]/7. xoneMx2,
     D3s[etaini] == 5./7. Exp[3. etaini]/9. (xoneMx2/(1. + (p/k)^2 + 2 p/k x) + xoneMx2/(1. + (p/k)^2 - 2 p/k x)) xoneMx2,
     D3s'[etaini] == 15./7. Exp[3. etaini]/9. (xoneMx2/(1. + (p/k)^2 + 2 p/k x) + xoneMx2/(1. + (p/k)^2 - 2 p/k x)) xoneMx2},
    {Dpk, Dpp, D2f, D2m, D3s}, {eta, etaini, etaev}, MaxSteps -> Infinity, AccuracyGoal -> 12, PrecisionGoal -> 12];
  Dpkv = Dpk[etaev] /. sol[[1]]; Dppv = Dpp[etaev] /. sol[[1]];
  D3v = D3s[etaev] /. sol[[1]]; dD3v = D3s'[etaev] /. sol[[1]];
  {D3v/(Dpkv Dppv Dppv), dD3v/(Dpkv Dppv Dppv)/(3 f0)}];

Print[""]; Print["=== Part 6: fused D3 (C3gamma3/C3gamma3f) at x=0.6,k=0.1,p=0.08 ==="];
